\newcommand*{\xmin}{-0.25}
\newcommand*{\xmax}{8.5}
\newcommand*{\ymin}{-1.25}
\newcommand*{\ymax}{1.25}

\begin{tikzpicture}
[
	closedpt/.style={circle,draw,fill,inner sep=0pt,minimum size=4pt},
	openpt/.style={circle,draw,inner sep=0pt,minimum size=4pt}
]

	% axes
	\draw[->,thick] (\xmin,0) -- (\xmax,0) node[right] {$t$};
	\draw[->,thick] (0,\ymin) -- (0,\ymax); % node[right] {$v(t)$};

	\draw[red,thick,domain=0:\xmax,smooth] plot (\x,{sin(deg(\x))})
		node[right] {$s(t)$};

	\draw[blue,thick,domain=0:\xmax,smooth] plot (\x,{cos(deg(\x))})
		node[right] {$v(t)$};

\end{tikzpicture}
